  Access the API Gateway
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

{{- if eq .Values.apiGateway.service.type "NodePort" }}
  export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[?(@.type=="InternalIP")].address}')
  curl http://${NODE_IP}:{{ .Values.apiGateway.service.nodePort }}/api/v1/gpus
  open http://${NODE_IP}:{{ .Values.apiGateway.service.nodePort }}/swagger/index.html
{{- else if eq .Values.apiGateway.service.type "LoadBalancer" }}
  export LB_IP=$(kubectl get svc {{ include "elastic-gpu-telemetry.fullname" . }}-api-gateway \
    -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
  curl http://${LB_IP}:{{ .Values.apiGateway.service.port }}/api/v1/gpus
  open http://${LB_IP}:{{ .Values.apiGateway.service.port }}/swagger/index.html
{{- else }}
  kubectl port-forward svc/{{ include "elastic-gpu-telemetry.fullname" . }}-api-gateway {{ .Values.apiGateway.service.port }}:{{ .Values.apiGateway.service.port }}
  curl http://localhost:{{ .Values.apiGateway.service.port }}/api/v1/gpus
  open http://localhost:{{ .Values.apiGateway.service.port }}/swagger/index.html
{{- end }}
